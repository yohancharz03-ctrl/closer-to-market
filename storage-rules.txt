  match /uploads/{uid}/{file} {
    allow read: if true;
    allow write: if request.auth != null && request.auth.uid == uid && request.resource.size < 15 * 1024 * 1024
      && request.resource.contentType.matches('image/.*|video/.*|application/pdf');
  } } }
